        // Generator Efek Suara (Menggunakan Web Audio API internal browser)
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        function mainkanSuara(jenis) {
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            
            if (jenis === 'benar') {
                osc.frequency.setValueAtTime(587.33, audioCtx.currentTime); // Nada D5
                osc.frequency.setValueAtTime(880, audioCtx.currentTime + 0.1);  // Nada A5
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.3);
            } else if (jenis === 'salah') {
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(150, audioCtx.currentTime);
                osc.frequency.linearRampToValueAtTime(100, audioCtx.currentTime + 0.2);
                gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.25);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.25);
            }
        }

function cekJawaban(pilihanUser, jawabanBenar) {
const semuaTombol = document.querySelectorAll('.btn-pilihan');
semuaTombol.forEach(tombol => tombol.disabled = true);
if (pilihanUser === jawabanBenar) {
document.getElementById(pil-${pilihanUser}).classList.add('betul');
mainkanSuara('benar');
skorSesiIni += 10;
totalSkorXP += 10;
document.getElementById('sesi-xp-info').innerText = +${skorSesiIni} XP Sesi Ini;
document.getElementById('total-xp').innerText = totalSkorXP;
} else {
document.getElementById(pil-${pilihanUser}).classList.add('salah');
document.getElementById(pil-${jawabanBenar}).classList.add('betul');
mainkanSuara('salah');
}
document.getElementById('btn-lanjut').classList.remove('hidden');
}

}
}
